parseStyleSheetCreate:
  (ast: t.File, source: string, warnings: string[], file: string) => Map<string, string[]>
  - On non-ObjectExpression arg → push warning "Unsupported StyleSheet.create argument: &lt;raw&gt; at &lt;file&gt;" + skip entry
  - On computed property key → skip that key (no warning per key — keep noise low)
  - varName resolved via path.parentPath?.node guarded by t.isVariableDeclarator
  - Returns empty Map for files with no StyleSheet.create calls

extractRNInlineStyle:
  (jsxElement: t.JSXElement, source: string) => Record<string, string | { raw: string }>
  - Body: `return extractInlineStyle(jsxElement, source)` — single-line delegation

extractNativeWindClassNames:
  (jsxElement: t.JSXElement, warnings: string[], file: string, line: number) => string[]
  - StringLiteral className → strip /(ios|android|web|native):/g globally, trim, split on /\s+/, filter Boolean
  - JSXExpressionContainer with TaggedTemplateExpression inside → push warning "NativeWind tw\` \` tagged template not supported — use className string at &lt;file&gt;:&lt;line&gt;" + return []
  - JSXExpressionContainer with other expr → return [] (no warning — silently unsupported until v1.3)
  - Missing className attribute → return []

flattenStyleArray:
  (node: t.JSXExpressionContainer, fileStyleIndex: Map<string, string[]>, source: string, warnings: string[], file: string) => string[]
  - expr not ArrayExpression → return []
  - elements iterated:
    * null sparse hole → continue (CHECK FIRST per Pitfall 4)
    * t.isNullLiteral / t.isBooleanLiteral → continue (no warning)
    * t.isStringLiteral → push value as key
    * t.isArrayExpression → warn "Nested array in style prop at &lt;file&gt; — skipped" + continue
    * t.isMemberExpression OR t.isLogicalExpression with t.isMemberExpression(el.right) → resolve via fileStyleIndex.get(varName); if found push all keys; if not push warning "StyleSheet var '&lt;name&gt;' not found in index at &lt;file&gt;"
    * t.isCallExpression → warn "Unsupported style array element (call expr): &lt;raw&gt; at &lt;file&gt;" + continue
</interfaces>
