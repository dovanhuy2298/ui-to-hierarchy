    /**
     * NextJsAdapter.detect — Phase 4, SPEC R5.
     *
     * Returns `true` only for projects that look like a Next.js App Router project:
     *   has any `next.config.{js,mjs,cjs,ts}` AND has `app/` (or `src/app/`).
     *
     * D-12 (no-throw): permission errors / ENOENT collapse to `false`. We never
     * `import()` config files — only `fs.access` them.
     */

    import { access } from "node:fs/promises";
    import { join } from "node:path";

    const NEXT_CONFIGS = [
      "next.config.js",
      "next.config.mjs",
      "next.config.cjs",
      "next.config.ts",
    ] as const;

    export async function detect(absRoot: string): Promise<boolean> {
      // 1) Any next.config.*?
      let hasConfig = false;
      for (const name of NEXT_CONFIGS) {
        if (await exists(join(absRoot, name))) {
          hasConfig = true;
          break;
        }
      }
      if (!hasConfig) return false;

      // 2) Any app/ or src/app/?
      if (await exists(join(absRoot, "app"))) return true;
      if (await exists(join(absRoot, "src", "app"))) return true;
      return false;
    }

    async function exists(p: string): Promise<boolean> {
      try {
        await access(p);
        return true;
      } catch {
        return false;
      }
    }
    ```

    No other exports. The function is async; use `node:path` `join` (OS-native is fine here — we only do `fs.access` checks, not URL emission).

    /**
     * NextJsAdapter.discoverEntries — Phase 4, SPEC R6.
     *
     * Enumerates App Router special files under `app/` (or `src/app/`) using
     * a single one-shot tinyglobby pass. Forward-slash normalized (D-08),
     * lex-sorted by explicit code-point comparator (Pitfall 4), private
     * folders excluded by glob `ignore` (D-09 single-source-of-truth).
     *
     * Also exports `resolveAppRoot` for plan 03's route-map.ts to reuse —
     * keeps the "app or src/app" probe in one place.
     */

    import { access } from "node:fs/promises";
    import { join } from "node:path";
    import { glob } from "tinyglobby";
    import { toForwardSlash } from "../../core/paths.js";

    const SPECIAL = "{page,layout,template,loading,error,not-found,default}";
    const EXTS = "{tsx,jsx,ts,js}";

    /**
     * Locate the App Router root (`<absRoot>/app` or `<absRoot>/src/app`).
     * Returns null when neither directory exists. Reused by plan 03's route-map.
     */
    export async function resolveAppRoot(absRoot: string): Promise<string | null> {
      for (const candidate of [join(absRoot, "app"), join(absRoot, "src", "app")]) {
        try {
          await access(candidate);
          return candidate;
        } catch {
          /* nope */
        }
      }
      return null;
    }

    export async function discoverEntries(absRoot: string): Promise<string[]> {
      const appRoot = await resolveAppRoot(absRoot);
      if (!appRoot) return [];

      const matches = await glob([`**/${SPECIAL}.${EXTS}`], {
        cwd: appRoot,
        absolute: true,
        ignore: ["**/_*/**", "**/node_modules/**"],
        dot: false,
      });

      return matches
        .map(toForwardSlash)
        .sort((a, b) => (a < b ? -1 : a > b ? 1 : 0));
    }
    ```

    Notes:
    - `tinyglobby`'s `glob` takes `string[]` patterns. The brace-expansion is handled inside the glob string itself.
    - Use `absolute: true` so we don't have to re-join paths after.
    - `dot: false` is the default but we spell it for clarity (avoids accidentally matching `.next/` build outputs if someone ever runs detect against a project that built once).
    - Do NOT pass `caseSensitiveMatch: false` — we want the literal allow-list.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| caller→`detect`/`discoverEntries` | `absRoot` is an absolute filesystem path passed by Phase 5 Analyzer (eventually from MCP tool input). Could be a path the user does not own. |
| filesystem→glob output | tinyglobby walks the filesystem under `cwd: appRoot`; output paths cross back into the adapter. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-04 | T (Tampering) / I (Info Disclosure) | `app/` is a symlink pointing outside the project root | mitigate | Glob `cwd: appRoot` confines enumeration to the app subtree. We do NOT call `fs.realpath` (per RESEARCH §"Security Domain"). If the symlink target is outside the project, we still only return paths under the link as-named — the user must approve the project root upstream. |
| T-04-05 | T (Tampering) | Symlink loop inside `app/` causes infinite walk | mitigate | tinyglobby has built-in symlink-loop protection (it tracks visited inodes). We further bound the walk by the special-file allow-list — the result set is naturally finite. |
| T-04-06 | I (Info Disclosure) | Glob accidentally walks `node_modules/` | mitigate | `ignore: ["**/node_modules/**"]` is hard-coded into every glob call. Plan 03 must replicate this exactly. |
| T-04-07 | E (Elevation of Privilege) | `next.config.ts` content gets `import()`ed during detect | accept | `detect()` only calls `fs.access` — never `import()`. Verified by grep on the file. Documented in 04-RESEARCH §"Security Domain". |
| T-04-08 | D (DoS) | Massive `app/` tree (>50k files) causes glob to hang | accept | v1 has no perf SLA (PROJECT.md). tinyglobby is well-tuned (~10x lighter than fast-glob). Phase 6 perf gate measures and decides whether to add limits. |
</threat_model>
