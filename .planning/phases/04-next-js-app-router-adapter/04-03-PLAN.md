    ```typescript
    /**
     * Pure regex-driven segment classifier (Phase 4, NEXT-02/03).
     *
     * Maps a Next.js App Router folder name to its SegmentKind variant.
     * No I/O, no throws — every input maps to exactly one variant.
     */

    export type SegmentKind =
      | { kind: "static"; name: string }
      | { kind: "dynamic"; param: string }
      | { kind: "catch-all"; param: string }
      | { kind: "optional-catch-all"; param: string }
      | { kind: "group"; label: string }
      | { kind: "parallel"; slot: string }
      | {
          kind: "intercepting";
          level: 0 | 1 | 2 | "root";
          targetSegment: string;
        }
      | { kind: "private"; name: string };

    // Order matters: more-specific patterns first.
    const RX_OPTIONAL_CATCH_ALL = /^\[\[\.\.\.([^\]]+)\]\]$/;
    const RX_CATCH_ALL = /^\[\.\.\.([^\]]+)\]$/;
    const RX_DYNAMIC = /^\[([^\]]+)\]$/;
    const RX_INTERCEPT_ROOT = /^\(\.\.\.\)(.+)$/;
    const RX_INTERCEPT_TWO = /^\(\.\.\)\(\.\.\)(.+)$/;
    const RX_INTERCEPT_ONE = /^\(\.\.\)(.+)$/;
    const RX_INTERCEPT_SAME = /^\(\.\)(.+)$/;
    const RX_GROUP = /^\(([^)]+)\)$/;
    const RX_PARALLEL = /^@(.+)$/;
    const RX_PRIVATE = /^_(.+)$/;

    export function classifySegment(folder: string): SegmentKind {
      let m: RegExpExecArray | null;

      if ((m = RX_OPTIONAL_CATCH_ALL.exec(folder)))
        return { kind: "optional-catch-all", param: m[1]! };
      if ((m = RX_CATCH_ALL.exec(folder))) return { kind: "catch-all", param: m[1]! };
      if ((m = RX_DYNAMIC.exec(folder))) return { kind: "dynamic", param: m[1]! };

      // Intercepting variants must be tested before plain group `(...)`
      // because (...)x and (..)x both start with paren-dot.
      if ((m = RX_INTERCEPT_ROOT.exec(folder)))
        return { kind: "intercepting", level: "root", targetSegment: m[1]! };
      if ((m = RX_INTERCEPT_TWO.exec(folder)))
        return { kind: "intercepting", level: 2, targetSegment: m[1]! };
      if ((m = RX_INTERCEPT_ONE.exec(folder)))
        return { kind: "intercepting", level: 1, targetSegment: m[1]! };
      if ((m = RX_INTERCEPT_SAME.exec(folder)))
        return { kind: "intercepting", level: 0, targetSegment: m[1]! };

      if ((m = RX_GROUP.exec(folder))) return { kind: "group", label: m[1]! };
      if ((m = RX_PARALLEL.exec(folder))) return { kind: "parallel", slot: m[1]! };
      if ((m = RX_PRIVATE.exec(folder))) return { kind: "private", name: m[1]! };

      return { kind: "static", name: folder };
    }

    export function extractParam(
      folderName: string,
    ): { name: string; kind: "single" | "catch-all" | "optional-catch-all" } | null {
      let m: RegExpExecArray | null;
      if ((m = RX_OPTIONAL_CATCH_ALL.exec(folderName)))
        return { name: m[1]!, kind: "optional-catch-all" };
      if ((m = RX_CATCH_ALL.exec(folderName))) return { name: m[1]!, kind: "catch-all" };
      if ((m = RX_DYNAMIC.exec(folderName))) return { name: m[1]!, kind: "single" };
      return null;
    }
    ```

    Verify ordering: optional-catch-all `[[...x]]` MUST be tried before catch-all `[...x]` MUST be tried before dynamic `[x]`. Intercepting `(...)x` MUST be tried before group `(x)` (because `(...)foo` would otherwise match `^\([^)]+\)$` for `(...)`). Verify the regex ordering above does this correctly.

    1. `splitRoute("/dashboard/settings")` → `["dashboard", "settings"]`. Reject leading `..` / `.` / empty / non-string with `matched: false`.
    2. Glob enumeration: same pattern as `discoverEntries` (D-06 / Pitfall 6 — one shot).
    3. For each glob hit, split its path-relative-to-appRoot into folder segments + filename. Walk the tree creating nodes as needed; classify each folder via `classifySegment`. Attach the file under the matched special-file key (page/layout/...).
    4. Walk the URL segments against the tree:
       - On a tree level, before descending, harvest any `(group)` children — their `layout.tsx`/specials feed into `entries` and we recurse into their children at the SAME URL position (the group is transparent). This may need to be transitive (group inside group).
       - Harvest any `@slot` siblings at each level — for each, recursively match the SAME remaining URL tail under the slot subtree and stuff results into `slots[slotName]`.
       - For each remaining URL token, find a matching child:
         - Exact static match → consume token, descend.
         - Dynamic `[slug]` child → consume token, set `params[slug] = token`, descend.
         - Catch-all `[...rest]` child → consume ALL remaining tokens, set `params[rest] = [...]`, descend (which must terminate at the catch-all node's `page.tsx`).
         - Optional catch-all `[[...opt]]` child → if tokens remain, behave like catch-all; if zero tokens remain AND the optional-catch-all child has a `page.tsx`, also a match with `params[opt] = []`.
         - Otherwise → no match, return `empty`.
       - When all URL tokens are consumed, the current node must have a `page.tsx`. If yes → success. If no but a `[[...opt]]` child exists with a `page.tsx`, that's also success (parent-path match for optional catch-all — Pitfall 3).
    5. Collect entries root-down: at each segment level visited (including transparent groups), append the node's special files in the documented within-segment order. The terminal page.tsx is appended last (research §"Layout Chain Semantics").
    6. Forward-slash normalize every emitted path (D-08) — prefer doing this once on the glob output before tree-build.

    ```typescript
     * NextJsAdapter.mapRouteToEntry — Phase 4, NEXT-01/02/03.
     *
     * One tinyglobby enumeration → build segment tree → walk URL segments.
     * The tree is a local variable in matchRoute, GC'd on return (ARCH-02).

     */

    import { glob } from "tinyglobby";
    import { toForwardSlash } from "../../core/paths.js";
    import type { RouteMatch } from "../types.js";
    import { resolveAppRoot } from "./discover.js";
    import { classifySegment, type SegmentKind } from "./segments.js";

    interface SegmentNode {
      segment: SegmentKind;
      children: Map<string, SegmentNode>;
      files: Partial<Record<
        "page" | "layout" | "template" | "loading" | "error" | "notFound" | "default",
        string
      >>;
      parallelSiblings: Map<string, SegmentNode>;
    }

    const EMPTY: RouteMatch = { matched: false, entries: [], params: {}, slots: {} };

    export async function matchRoute(absRoot: string, route: string): Promise<RouteMatch> {
      // D-12: malformed input
      if (typeof route !== "string" || !route.startsWith("/")) {
        return cloneEmpty();
      }
      const segments = route.split("/").filter(Boolean);
      // Reject path-traversal attempts: `.` and `..` are not legal route tokens
      if (segments.some((s) => s === "." || s === "..")) return cloneEmpty();

      const appRoot = await resolveAppRoot(absRoot);
      if (!appRoot) return cloneEmpty();
      const fwdAppRoot = toForwardSlash(appRoot);

      const files = await glob(
        ["**/{page,layout,template,loading,error,not-found,default}.{tsx,jsx,ts,js}"],
        { cwd: appRoot, absolute: true, ignore: ["**/_*/**", "**/node_modules/**"], dot: false },
      );
      const fwdFiles = files.map(toForwardSlash);

      const tree = buildTree(fwdFiles, fwdAppRoot);
      return walk(tree, segments);
    }

    function cloneEmpty(): RouteMatch {
      return { matched: false, entries: [], params: {}, slots: {} };
    }

    // ... buildTree, walk, harvestSpecials, matchOptionalCatchAll, etc.
    ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-09 | T (Tampering) | Route input `/../../etc/passwd` attempts to escape `app/` | mitigate | Reject any URL segment equal to `.` or `..` at the top of `matchRoute`; return `cloneEmpty()`. Asserted by D-12 test "returns matched:false on path-traversal route /../etc". Even if rejection failed, the tree is built from glob output `cwd: appRoot`, so the matched entries can only point to files under `app/` — there is no path that lets the walker emit a path outside the project. |
| T-04-10 | I (Info Disclosure) | Glob walks into `_private` even though we said not to | mitigate | Two layers: (1) glob `ignore: ["**/_*/**", "**/node_modules/**"]` — same as plan 02 (defense-in-depth, both globs share this list); (2) classifier marks any `_*` folder as `{ kind: "private" }` and the walker skips private children. Asserted by NEXT-02 test "_internal is excluded from any RouteMatch". |
| T-04-11 | T (Tampering) | Symlink loop in `app/` causes infinite walk | mitigate | tinyglobby has built-in symlink-loop protection. The downstream tree-build is finite by construction (each file produces at most one path of segments). |
| T-04-12 | D (DoS) | Route with 10000 segments causes deep recursion | accept | The walker is segment-bounded by the URL length. v1 has no SLA. If observed in practice, a 256-segment cap can be added in a follow-up. |
