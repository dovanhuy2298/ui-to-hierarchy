**File 1 — `src/core/resolver/relative.ts`:**

Pure utility — given an absolute base path (no extension), probe disk and return the first existing file or null.

```typescript
import { existsSync, statSync } from "node:fs";
import path from "node:path";
import { toForwardSlash } from "../paths.js";

const EXT_ORDER = [".ts", ".tsx", ".js", ".jsx"] as const;
const INDEX_ORDER = ["/index.ts", "/index.tsx", "/index.js", "/index.jsx"] as const;

/**
 * Probe a base path with D-13 extension order.
 * Returns the first existing forward-slash absolute file path, or null.
 */
export function probeFile(basePath: string): string | null {
  // 1. Exact path (already has extension, e.g., "./styles.css" — skip if not a TS/JS source)
  if (existsSync(basePath) && safeStatIsFile(basePath)) {
    return toForwardSlash(path.resolve(basePath));
  }
  // 2. <base>.<ext>
  for (const ext of EXT_ORDER) {
    const candidate = basePath + ext;
    if (existsSync(candidate) && safeStatIsFile(candidate)) {
      return toForwardSlash(path.resolve(candidate));
    }
  }
  // 3. <base>/index.<ext>
  for (const idx of INDEX_ORDER) {
    const candidate = basePath + idx;
    if (existsSync(candidate) && safeStatIsFile(candidate)) {
      return toForwardSlash(path.resolve(candidate));
    }
  }
  return null;
}

function safeStatIsFile(p: string): boolean {
  try { return statSync(p).isFile(); } catch { return false; }
}

/** Resolve `./foo` or `../bar` from a fromFile to an absolute base path (no extension probe). */
export function joinRelative(fromFile: string, specifier: string): string {
  return path.resolve(path.dirname(fromFile), specifier);
}
```

```typescript
import { getTsconfig, createPathsMatcher, type TsconfigResult } from "get-tsconfig";
import type { ParseContext } from "../../adapters/types.js";

/**
 * Load tsconfig once per ParseContext (D-01). `getTsconfig` walks the `extends`
 * chain transparently. ctx.tsconfig === null indicates "tried and didn't find".
 */
export function loadTsconfigOnce(ctx: ParseContext): TsconfigResult | null {
  // We use null as both "not loaded" and "no tsconfig found". To distinguish,
  // attach a sentinel on first call. ParseContext.tsconfig is set by the caller
  // (NextJsAdapter); only resolve here if undefined was passed.
  if (ctx.tsconfig !== null) return ctx.tsconfig;
  const tsconfig = getTsconfig(ctx.resolvedRoot);
  ctx.tsconfig = tsconfig;
  return tsconfig;
}

/**
 * Build the paths matcher once per ParseContext. Cached on a hidden symbol-keyed
 * field of ctx (TypeScript-only — JS-level Map<ParseContext, ...> would defeat GC).
 */
const matcherCache = new WeakMap<ParseContext, ((spec: string) => string[]) | null>();

export function getPathsMatcher(ctx: ParseContext): ((specifier: string) => string[]) | null {
  if (matcherCache.has(ctx)) return matcherCache.get(ctx)!;
  const tsconfig = loadTsconfigOnce(ctx);
  const matcher = tsconfig ? createPathsMatcher(tsconfig) : null;
  matcherCache.set(ctx, matcher);
  return matcher;
}
```

```typescript
/**
 * Detect node_modules boundary on a forward-slash absolute path.
 * Returns the package name (scoped @org/pkg or unscoped) or null.
 */
export function detectNodeModules(absForwardSlash: string): string | null {
  const idx = absForwardSlash.lastIndexOf("/node_modules/");
  if (idx === -1) return null;
  const after = absForwardSlash.slice(idx + "/node_modules/".length);
  const segs = after.split("/");
  if (segs.length === 0 || segs[0] === "") return null;
  if (segs[0].startsWith("@") && segs.length >= 2) return `${segs[0]}/${segs[1]}`;
  return segs[0];
}

/**
 * Extract package name from an unresolved bare specifier (e.g. "react", "@radix-ui/react-dialog/foo").
 * Returns null for relative or alias specifiers.
 */
export function packageNameFromSpecifier(specifier: string): string | null {
  if (specifier.startsWith(".") || specifier.startsWith("/") || specifier.startsWith("@/") || specifier.startsWith("~/") || specifier.startsWith("#")) return null;
  const segs = specifier.split("/");
  if (segs[0].startsWith("@") && segs.length >= 2) return `${segs[0]}/${segs[1]}`;
  return segs[0];
}
```

**File 1 — `src/core/resolver/barrel.ts`:**

Implement `chaseBarrel(ctx, startFile, importedName, visited)` per RESEARCH Pattern 4:

```typescript
import * as t from "@babel/types";
import { traverse } from "../babel-shim.js";
import { parseFile } from "../parser/index.js";
import type { ParseContext, ResolveResult } from "../../adapters/types.js";

/**
 * Chase a named export through `export { Foo } from "..."` and `export * from "..."`
 * declarations until the local declaration is found, the chain hits node_modules,
 * or a cycle is detected.
 *
 * `resolveSpecifier` is injected to break the import cycle with index.ts.
 */
export function chaseBarrel(
  ctx: ParseContext,
  startFile: string,
  importedName: string,
  resolveSpecifier: (ctx: ParseContext, fromFile: string, specifier: string) => ResolveResult,
  visited: Set<string> = new Set(),
): ResolveResult {
  if (visited.has(startFile)) {
    return { ok: false, kind: "cycle", chain: [...visited, startFile] };
  }
  visited.add(startFile);

  const parsed = parseFile(ctx, startFile);
  if (parsed.kind === "error") {
    return { ok: false, kind: "not-found", specifier: importedName, tried: [startFile] };
  }

  let foundLocal = false;
  let reExportFrom: { source: string; renamed: string } | null = null;
  const starExports: string[] = [];

  traverse(parsed.ast, {
    FunctionDeclaration(p) {
      if (p.node.id?.name === importedName) foundLocal = true;
    },
    VariableDeclarator(p) {
      if (t.isIdentifier(p.node.id) && p.node.id.name === importedName) foundLocal = true;
    },
    ClassDeclaration(p) {
      if (p.node.id?.name === importedName) foundLocal = true;
    },
    ExportNamedDeclaration(p) {
      if (!p.node.source) return;
      for (const spec of p.node.specifiers) {
        if (t.isExportSpecifier(spec)) {
          const exportedName = t.isIdentifier(spec.exported) ? spec.exported.name : spec.exported.value;
          if (exportedName === importedName) {
            const renamed = t.isIdentifier(spec.local) ? spec.local.name : importedName;
            reExportFrom = { source: p.node.source.value, renamed };
          }
        }
      }
    },
    ExportAllDeclaration(p) {
      starExports.push(p.node.source.value);
    },
  });

  if (foundLocal) return { ok: true, kind: "local", absolutePath: startFile };
  if (reExportFrom) {
    const next = resolveSpecifier(ctx, startFile, reExportFrom.source);
    if (!next.ok) return next;
    if (next.kind === "external") return next;
    return chaseBarrel(ctx, next.absolutePath, reExportFrom.renamed, resolveSpecifier, visited);
  }
  for (const starSource of starExports) {
    const next = resolveSpecifier(ctx, startFile, starSource);
    if (!next.ok) continue;
    if (next.kind === "external") continue;
    // Fork visited per branch so sibling stars don't poison each other,
    // but propagate cycles upward.
    const sub = chaseBarrel(ctx, next.absolutePath, importedName, resolveSpecifier, new Set(visited));
    if (sub.ok) return sub;
    if (!sub.ok && sub.kind === "cycle") return sub;
  }
  return { ok: false, kind: "not-found", specifier: importedName, tried: [startFile] };
}
```

**File 2 — `src/core/resolver/index.ts`:**

The entry point `resolveModule(ctx, fromFile, specifier, importedName)`:

1. Compute cache key `${fromFile}::${specifier}::${importedName}` (D-03). Return cached if present.
2. Quick-resolve to a file (specifier-only, ignoring importedName for the moment) via:
   - tsconfig paths (`getPathsMatcher(ctx)(specifier)` → array of base paths; iterate; for each, `probeFile`).
   - Bare specifier with no alias hit → check `packageNameFromSpecifier`; if defined, return `{ ok: true, kind: "external", packageName }`.
   - Relative specifier → `joinRelative(fromFile, specifier)` then `probeFile`.
3. After file found:
   - Forward-slash + check `detectNodeModules(absPath)` → if matched, return `{ ok: true, kind: "external", packageName }`.
   - Parse the file and look for a local declaration of `importedName`. If found → `{ ok: true, kind: "local", absolutePath }`.
   - If not found → call `chaseBarrel(ctx, absPath, importedName, resolveSpecifierForBarrel)` where `resolveSpecifierForBarrel` is a closure that recursively calls a SPECIFIER-ONLY resolution (no importedName) — pass `"*"` as the importedName placeholder, but the chase function actually only needs the specifier→file step. Implementation detail: extract the specifier→file logic into a private helper `resolveSpecifierToFile(ctx, fromFile, specifier): ResolveResult` and pass that to `chaseBarrel`.
4. Cache the final result and return.

```typescript
import path from "node:path";
import * as t from "@babel/types";
import { traverse } from "../babel-shim.js";
import { parseFile } from "../parser/index.js";
import { toForwardSlash } from "../paths.js";
import type { ParseContext, ResolveResult } from "../../adapters/types.js";
import { joinRelative, probeFile } from "./relative.js";
import { getPathsMatcher } from "./tsconfig.js";
import { detectNodeModules, packageNameFromSpecifier } from "./node-modules.js";
import { chaseBarrel } from "./barrel.js";

export function resolveModule(
  ctx: ParseContext,
  fromFile: string,
  specifier: string,
  importedName: string,
): ResolveResult {
  const key = `${fromFile}::${specifier}::${importedName}`;
  const cached = ctx.resolverCache.get(key);
  if (cached) return cached;

  const result = doResolve(ctx, fromFile, specifier, importedName);
  ctx.resolverCache.set(key, result);
  return result;
}

function resolveSpecifierToFile(ctx: ParseContext, fromFile: string, specifier: string): ResolveResult {
  // Tsconfig paths
  const matcher = getPathsMatcher(ctx);
  if (matcher) {
    const candidates = matcher(specifier);
    const tried: string[] = [];
    for (const cand of candidates) {
      tried.push(cand);
      const hit = probeFile(cand);
      if (hit) {
        const fwd = toForwardSlash(hit);
        const ext = detectNodeModules(fwd);
        return ext ? { ok: true, kind: "external", packageName: ext } : { ok: true, kind: "local", absolutePath: fwd };
      }
    }
    if (candidates.length > 0) {
      // Path matched but no file existed — fall through to bare/relative attempt.
    }
  }
  // Relative
  if (specifier.startsWith(".") || specifier.startsWith("/")) {
    const base = joinRelative(fromFile, specifier);
    const hit = probeFile(base);
    if (hit) {
      const fwd = toForwardSlash(hit);
      const ext = detectNodeModules(fwd);
      return ext ? { ok: true, kind: "external", packageName: ext } : { ok: true, kind: "local", absolutePath: fwd };
    }
    return { ok: false, kind: "not-found", specifier, tried: [base] };
  }
  // Bare specifier — node_modules without resolution (D-12 external).
  const pkg = packageNameFromSpecifier(specifier);
  if (pkg) return { ok: true, kind: "external", packageName: pkg };
  return { ok: false, kind: "not-found", specifier, tried: [] };
}

function doResolve(ctx: ParseContext, fromFile: string, specifier: string, importedName: string): ResolveResult {
  const fileResult = resolveSpecifierToFile(ctx, fromFile, specifier);
  if (!fileResult.ok) return fileResult;
  if (fileResult.kind === "external") return fileResult;
  // Local file found — does it declare importedName itself, or is it a barrel?
  const parsed = parseFile(ctx, fileResult.absolutePath);
  if (parsed.kind === "error") {
    return { ok: false, kind: "not-found", specifier, tried: [fileResult.absolutePath] };
  }
  let foundLocal = false;
  traverse(parsed.ast, {
    FunctionDeclaration(p) { if (p.node.id?.name === importedName) foundLocal = true; },
    VariableDeclarator(p) { if (t.isIdentifier(p.node.id) && p.node.id.name === importedName) foundLocal = true; },
    ClassDeclaration(p) { if (p.node.id?.name === importedName) foundLocal = true; },
  });
  if (foundLocal) return fileResult;
  // Barrel chase
  return chaseBarrel(ctx, fileResult.absolutePath, importedName, resolveSpecifierToFile);
}
```

```typescript
import { describe, it, expect } from "vitest";
import path from "node:path";
import { resolveModule } from "../../../src/core/resolver/index.js";
import type { ParseContext } from "../../../src/adapters/types.js";

function ctxFor(rootRel: string): ParseContext {
  return {
    resolvedRoot: path.resolve(rootRel),
    tsconfig: null,
    astCache: new Map(),
    resolverCache: new Map(),
    warnings: [],
  };
}

describe("PARSE-03 tsconfig paths", () => {
  it("resolves @/* alias on shadcn-barrel fixture", () => {
    const ctx = ctxFor("test/fixtures/parser/resolver/shadcn-barrel");
    const fromFile = path.resolve("test/fixtures/parser/resolver/shadcn-barrel/src/page.tsx");
    const r = resolveModule(ctx, fromFile, "@/components/ui", "Button");
    expect(r.ok).toBe(true);
    if (r.ok && r.kind === "local") {
      expect(r.absolutePath.endsWith("/components/ui/button.tsx")).toBe(true);
      expect(r.absolutePath.includes("\\")).toBe(false);
    }
  });

  it("resolves multi-target paths first-wins (Foo from src/, Bar from lib/)", () => {
    const ctx = ctxFor("test/fixtures/parser/resolver/multi-target");
    const fromFile = path.resolve("test/fixtures/parser/resolver/multi-target/page.tsx");
    const foo = resolveModule(ctx, fromFile, "@/components/Foo", "Foo");
    const bar = resolveModule(ctx, fromFile, "@/components/Bar", "Bar");
    expect(foo.ok && foo.kind === "local" && foo.absolutePath.endsWith("/src/components/Foo.tsx")).toBe(true);
    expect(bar.ok && bar.kind === "local" && bar.absolutePath.endsWith("/lib/components/Bar.tsx")).toBe(true);
  });

  it("honors tsconfig.extends chain", () => {
    const ctx = ctxFor("test/fixtures/parser/resolver/extends-chain");
    const fromFile = path.resolve("test/fixtures/parser/resolver/extends-chain/src/x.ts");
    const r = resolveModule(ctx, fromFile, "#config/x", "X");
    expect(r.ok).toBe(true);
    if (r.ok && r.kind === "local") expect(r.absolutePath.endsWith("/src/x.ts")).toBe(true);
  });

  it("D-03 cache: identical key returns same object on re-entry", () => {
    const ctx = ctxFor("test/fixtures/parser/resolver/shadcn-barrel");
    const fromFile = path.resolve("test/fixtures/parser/resolver/shadcn-barrel/src/page.tsx");
    const a = resolveModule(ctx, fromFile, "@/components/ui", "Button");
    const b = resolveModule(ctx, fromFile, "@/components/ui", "Button");

  });
});
```

```typescript
import { describe, it, expect } from "vitest";
import path from "node:path";
import { resolveModule } from "../../../src/core/resolver/index.js";
import type { ParseContext } from "../../../src/adapters/types.js";

function ctxFor(rootRel: string): ParseContext {
  return { resolvedRoot: path.resolve(rootRel), tsconfig: null, astCache: new Map(), resolverCache: new Map(), warnings: [] };
}

describe("PARSE-02 barrel chase", () => {
  it("chases shadcn-style barrel to source file", () => {
    const ctx = ctxFor("test/fixtures/parser/resolver/shadcn-barrel");
    const fromFile = path.resolve("test/fixtures/parser/resolver/shadcn-barrel/src/page.tsx");
    const r = resolveModule(ctx, fromFile, "@/components/ui", "Button");
    expect(r.ok && r.kind === "local").toBe(true);
    if (r.ok && r.kind === "local") expect(r.absolutePath.endsWith("/components/ui/button.tsx")).toBe(true);
  });

  it("emits cycle for a → b → a re-export cycle (no stack overflow)", () => {
    const ctx = ctxFor("test/fixtures/parser/resolver/barrel-cycle");
    const fromFile = path.resolve("test/fixtures/parser/resolver/barrel-cycle/src/page.tsx");
    const r = resolveModule(ctx, fromFile, "@/a", "Thing");
    expect(r.ok).toBe(false);
    if (!r.ok) {
      expect(r.kind).toBe("cycle");
      if (r.kind === "cycle") {
        expect(r.chain.length).toBeGreaterThanOrEqual(2);
      }
    }
  });

  it("never throws on missing import — returns not-found", () => {
    const ctx = ctxFor("test/fixtures/parser/resolver/shadcn-barrel");
    const fromFile = path.resolve("test/fixtures/parser/resolver/shadcn-barrel/src/page.tsx");
    expect(() => resolveModule(ctx, fromFile, "@/does/not/exist", "Whatever")).not.toThrow();
    const r = resolveModule(ctx, fromFile, "@/does/not/exist", "Whatever");
    expect(r.ok).toBe(false);
    if (!r.ok) expect(r.kind).toBe("not-found");
  });
});
```
