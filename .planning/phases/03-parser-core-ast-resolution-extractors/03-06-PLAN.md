```typescript
import * as t from "@babel/types";
import { traverse } from "../../core/babel-shim.js";
import { parseFile } from "../../core/parser/index.js";
import { resolveModule as coreResolveModule } from "../../core/resolver/index.js";
import { walkRenderFlow } from "../../core/render-flow/index.js";
import { discoverComponents, type DiscoveredComponent } from "../../core/render-flow/component-detect.js";
import { collectStyleSignals } from "../../core/extractors/index.js";
import { toForwardSlash } from "../../core/paths.js";
import type { FrameworkAdapter } from "../FrameworkAdapter.js";
import type {
  ComponentDefinition, ParseContext, PropSignature, RenderNode, ResolveResult,
} from "../types.js";

const NOT_IMPLEMENTED = "not implemented in Phase 3";

export const NextJsAdapter: FrameworkAdapter = {
  detect(): never { throw new Error(NOT_IMPLEMENTED); },
  discoverEntries(): never { throw new Error(NOT_IMPLEMENTED); },
  mapRouteToEntry(): never { throw new Error(NOT_IMPLEMENTED); },

  resolveModule(ctx, fromFile, specifier, importedName): ResolveResult {
    return coreResolveModule(ctx, fromFile, specifier, importedName);
  },

  extractComponents(ctx, entryFiles, opts = {}): ComponentDefinition[] {
    const out: ComponentDefinition[] = [];
    for (const absPath of entryFiles) {
      const fwdFile = toForwardSlash(absPath);
      const parsed = parseFile(ctx, absPath);
      if (parsed.kind === "error") {
        // SPEC R1: file-level parse error becomes a kind:"error" component
        out.push({
          name: "<parse-error>",
          file: fwdFile,
          line: parsed.line,
          kind: "function",
          wrappers: [],
          props: [],
          textContent: [],
          renderFlow: { kind: "error", message: parsed.message, file: fwdFile, line: parsed.line },
          classNames: [],
          inlineStyles: {},
          cssModuleRefs: [],
          styledTemplates: [],
        });
        continue;
      }
      const components = discoverComponents(parsed.ast);
      for (const comp of components) {
        out.push(buildComponentDefinition(ctx, comp, parsed.ast, parsed.source, fwdFile, opts));
      }
    }
    return out;
  },
};

function buildComponentDefinition(
  ctx: ParseContext,
  comp: DiscoveredComponent,
  ast: t.File,
  source: string,
  file: string,
  opts: { fullClasses?: boolean },
): ComponentDefinition {
  // 1. renderFlow
  const renderFlow = walkRenderFlow(comp.body, source, file);

  // 2. JSXElement[] within this component's body for per-element style signals
  const jsxElements = collectJsxElements(comp.body);

  // 3. Style signals (Tailwind tokens are filtered by opts.fullClasses; the rest are full)
  const styleSignals = collectStyleSignals(ast, jsxElements, source, file, opts);

  // 4. Props extraction (D-06, D-07) — minimal: function param 0 destructure, or single positional
  const props = extractProps(comp.body, source);

  // 5. textContent — string literals in JSXText
  const textContent = collectTextContent(renderFlow);

  return {
    name: comp.name,
    file,
    line: comp.declarationLine,
    kind: comp.kind,
    wrappers: comp.wrappers,
    props,
    textContent,
    renderFlow,
    classNames: styleSignals.classNames,
    inlineStyles: styleSignals.inlineStyles,
    cssModuleRefs: styleSignals.cssModuleRefs,
    styledTemplates: styleSignals.styledTemplates,
  };
}

function collectJsxElements(node: t.Node): t.JSXElement[] {
  const out: t.JSXElement[] = [];
  // We need traverse to descend; build a synthetic file-ish scope
  // Workaround: traverse expects a Program/File at top — wrap.
  const fakeFile: t.File = t.file(t.program([t.expressionStatement(node as t.Expression)]));
  // The above only works if node is an Expression. For ClassMethod / FunctionDeclaration we walk children manually.
  if (t.isJSXElement(node)) out.push(node);
  walkAst(node, (n) => {
    if (t.isJSXElement(n)) out.push(n);
  });
  return out;
}

/** Minimal manual walker (avoids the synthetic-file gymnastics). */
function walkAst(node: t.Node | null | undefined, visit: (n: t.Node) => void): void {
  if (!node) return;
  visit(node);
  for (const key of Object.keys(node) as (keyof typeof node)[]) {
    if (key === "loc" || key === "start" || key === "end" || key === "range" || key === "leadingComments" || key === "trailingComments" || key === "innerComments") continue;
    const child = (node as any)[key];
    if (Array.isArray(child)) {
      for (const c of child) {
        if (c && typeof c === "object" && typeof (c as any).type === "string") walkAst(c, visit);
      }
    } else if (child && typeof child === "object" && typeof (child as any).type === "string") {
      walkAst(child, visit);
    }
  }
}

function extractProps(body: t.Node, source: string): PropSignature[] {
  // body is FunctionDeclaration | FunctionExpression | ArrowFunctionExpression | ClassMethod | Identifier (HOC arg).
  let fn: t.FunctionDeclaration | t.FunctionExpression | t.ArrowFunctionExpression | null = null;
  if (t.isFunctionDeclaration(body) || t.isFunctionExpression(body) || t.isArrowFunctionExpression(body)) fn = body;
  if (!fn || fn.params.length === 0) return [];
  const param = fn.params[0];

  // Capture the type slice once — used for every destructured field per D-07
  let typeSlice = "";
  let typeOptional = false;
  if (t.isIdentifier(param) && param.typeAnnotation && t.isTSTypeAnnotation(param.typeAnnotation)) {
    typeSlice = sliceSource(source, param.typeAnnotation.typeAnnotation);
    typeOptional = !!param.optional;
  } else if (t.isObjectPattern(param) && param.typeAnnotation && t.isTSTypeAnnotation(param.typeAnnotation)) {
    typeSlice = sliceSource(source, param.typeAnnotation.typeAnnotation);
  } else if (t.isAssignmentPattern(param) && t.isObjectPattern(param.left) && param.left.typeAnnotation && t.isTSTypeAnnotation(param.left.typeAnnotation)) {
    typeSlice = sliceSource(source, param.left.typeAnnotation.typeAnnotation);
  }

  // Pure positional: function Card(props: Props) → single synthetic `props`
  if (t.isIdentifier(param)) {
    return [{ name: param.name, typeSlice, optional: typeOptional }];
  }

  // Destructure: { a, b: alias, ...rest } — emit a, alias, rest with shared typeSlice
  const objPat = t.isObjectPattern(param) ? param : (t.isAssignmentPattern(param) && t.isObjectPattern(param.left) ? param.left : null);
  if (!objPat) return [];
  const props: PropSignature[] = [];
  for (const p of objPat.properties) {
    if (t.isObjectProperty(p)) {
      // value is the local name (or a nested pattern)
      let localName: string | null = null;
      let optional = false;
      if (t.isIdentifier(p.value)) localName = p.value.name;
      else if (t.isAssignmentPattern(p.value) && t.isIdentifier(p.value.left)) {
        localName = p.value.left.name;
        optional = true; // has a default value
      }
      if (localName) props.push({ name: localName, typeSlice, optional });
    } else if (t.isRestElement(p)) {
      if (t.isIdentifier(p.argument)) props.push({ name: p.argument.name, typeSlice, optional: false });
    }
  }
  return props;
}

function collectTextContent(node: RenderNode | null): string[] {
  if (!node) return [];
  const out: string[] = [];
  visit(node);
  return out;
  function visit(n: RenderNode | null) {
    if (!n) return;
    if (n.kind === "text") out.push(n.value);
    if (n.kind === "jsx" || n.kind === "fragment") for (const c of n.children) visit(c);
    if (n.kind === "branch") { visit(n.thenBranch); visit(n.elseBranch); }
    if (n.kind === "list") visit(n.item);
  }
}

function sliceSource(source: string, node: t.Node): string {
  const start = node.start ?? 0;
  const end = node.end ?? start;
  return source.slice(start, end);
}
```

```typescript
import { describe, it, expect } from "vitest";
import path from "node:path";
import { NextJsAdapter } from "../../../src/adapters/next/NextJsAdapter.js";
import type { ParseContext } from "../../../src/adapters/types.js";

const FIXTURE = path.resolve("test/fixtures/parser/extractors/kitchen-sink.tsx");

function ctx(): ParseContext {
  return { resolvedRoot: path.resolve("."), tsconfig: null, astCache: new Map(), resolverCache: new Map(), warnings: [] };

describe("Kitchen-sink end-to-end (SPEC R5 acceptance)", () => {
  it("produces a ComponentDefinition with all four style fields populated", () => {
    const c = ctx();
    const comps = NextJsAdapter.extractComponents(c, [FIXTURE]);
    const ks = comps.find((x) => x.name === "KS");
    expect(ks).toBeDefined();
    if (!ks) return;

    // Tailwind classNames — has at least flex, p-4 (layout), and a `raw` entry from `cn(...)` non-literal arg
    const literals = ks.classNames.filter((c) => c.kind === "literal").map((c) => c.value);
    expect(literals).toContain("flex");
    expect(literals).toContain("p-4");
    // Default fullClasses=false: text-red-500 should be filtered out
    expect(literals).not.toContain("text-red-500");

    // Inline style
    expect(ks.inlineStyles.margin).toBe("8");

    // CSS Modules ref
    expect(ks.cssModuleRefs.some((r) => r.binding === "styles" && r.key === "label")).toBe(true);

    // styled-components template
    expect(ks.styledTemplates.some((t) => t.tag === "section" && t.body.includes("border:") && t.body.includes("{?}"))).toBe(true);

    // renderFlow contains the section JSX with children
    expect(ks.renderFlow.kind).toBe("jsx");
  });

  it("fullClasses=true reveals non-layout Tailwind tokens", () => {
    const c = ctx();
    const comps = NextJsAdapter.extractComponents(c, [FIXTURE], { fullClasses: true });
    const ks = comps.find((x) => x.name === "KS");
    expect(ks).toBeDefined();
    if (!ks) return;
    const literals = ks.classNames.filter((c) => c.kind === "literal").map((c) => c.value);
    expect(literals).toContain("text-red-500");
    expect(literals).toContain("flex");
  });
});
```
